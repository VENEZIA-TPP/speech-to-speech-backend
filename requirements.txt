# Voice activity detection
# Silero VAD ships as ONNX weights vendored under app/pipeline/weights/, so the
# only runtime dependency is the ONNX Runtime (~19 MB) plus numpy. The
# silero-vad package on PyPI is NOT used: it hard-requires torch + torchaudio
# (~1 GB) as base dependencies to serve a 1.3 MB model, with ONNX only as an
# extra.
# The floor keeps a careless upgrade out of the releases onnx-asr excludes
# (1.24.1, 1.25.*, 1.26.0 - all below it) and matches the version this repo
# actually runs and tests against.
onnxruntime>=1.28

# Speech recognition
# Parakeet TDT runs on onnx-asr, whose only base dependency is numpy:
# onnxruntime is already here for the VAD and huggingface_hub for the MT
# artifacts, so this line adds ~4 MB of loader code and no new runtime. The
# ONNX weights themselves are a third-party export pinned by commit in
# asr_service.py. ASR_MODEL=stub imports none of it: the loader imports
# inside itself.
onnx-asr

# Machine translation
# OPUS-MT runs on CTranslate2, whose only dependencies are numpy and pyyaml -
# no torch on the inference path. The published conversions ship source.spm and
# target.spm, so tokenization needs sentencepiece alone rather than
# transformers, and huggingface_hub is what fetches the pinned revisions.
# MT_MODEL=stub imports none of the three: the loader imports inside itself.
ctranslate2
sentencepiece
huggingface_hub
